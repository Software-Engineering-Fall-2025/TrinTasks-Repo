# TrinTasks Mac App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Convert the TrinTasks Chrome extension into a native macOS menu bar app using Swift/SwiftUI that preserves the neo-brutalism design.

**Architecture:** `AppDelegate` owns an `NSStatusItem` (menu bar icon) and an `NSPanel` (floating window). Services layer handles iCal fetching, background refresh, and notifications. SwiftUI views consume an `ObservableObject` store. The existing JS iCal parser is ported to Swift rather than using a third-party library, preserving Trinity-specific parsing logic.

**Tech Stack:** Swift 5.9+, SwiftUI, AppKit (NSStatusItem/NSPanel), XCTest, UNUserNotificationCenter, NSBackgroundActivityScheduler, create-dmg (distribution)

> **Note on themes:** The extension has 6 themes (fern, ocean, sunset, slate, orchid, midnight) plus a "glass" UI style overlay. The spec mentioned "Serenity" — this does not exist in the extension; "glass" is the equivalent. This plan implements the 6 themes + glass style.

```swift
import Foundation

final class StorageService {
    private let defaults: UserDefaults
    private let settingsKey = "appSettings"
    private let metadataPrefix = "metadata."

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
    }

    func saveSettings(_ settings: AppSettings) {
        if let data = try? JSONEncoder().encode(settings) {
            defaults.set(data, forKey: settingsKey)
        }
    }

    func loadSettings() -> AppSettings {
        guard let data = defaults.data(forKey: settingsKey),
              let settings = try? JSONDecoder().decode(AppSettings.self, from: data)
        else { return AppSettings() }
        return settings
    }

    func saveMetadata(_ metadata: EventMetadata, forUID uid: String) {
        if let data = try? JSONEncoder().encode(metadata) {
            defaults.set(data, forKey: metadataPrefix + uid)
        }
    }

    func loadMetadata(forUID uid: String) -> EventMetadata {
        guard let data = defaults.data(forKey: metadataPrefix + uid),
              let metadata = try? JSONDecoder().decode(EventMetadata.self, from: data)
        else { return EventMetadata() }
        return metadata
    }

    func merge(events: [CalendarEvent]) -> [Assignment] {
        return events.map { event in
            Assignment(event: event, metadata: loadMetadata(forUID: event.uid))
        }
    }
}
```

Create `TrinTasksApp/Services/ICalService.swift`:

```swift
import Foundation

final class ICalService {

    // MARK: - Public API

    static func parse(icsString: String) -> [CalendarEvent] {
        let eventBlocks = extractEventBlocks(from: icsString)
        return eventBlocks.compactMap { parseEvent($0) }
    }

    static func fetch(url: URL) async throws -> [CalendarEvent] {
        let (data, _) = try await URLSession.shared.data(from: url)
        guard let icsString = String(data: data, encoding: .utf8) else {
            throw ICalError.invalidEncoding
        }
        return parse(icsString: icsString)
    }

    // MARK: - Parsing

    private static func extractEventBlocks(from ics: String) -> [String] {
        var blocks: [String] = []
        let pattern = "BEGIN:VEVENT([\\s\\S]*?)END:VEVENT"
        guard let regex = try? NSRegularExpression(pattern: pattern) else { return [] }
        let range = NSRange(ics.startIndex..., in: ics)
        let matches = regex.matches(in: ics, range: range)
        for match in matches {
            if let range = Range(match.range(at: 1), in: ics) {
                blocks.append(String(ics[range]))
            }
        }
        return blocks
    }

    private static func parseEvent(_ block: String) -> CalendarEvent? {
        let uid = extractField("UID", from: block) ?? UUID().uuidString
        var title = extractFoldedField("SUMMARY", from: block) ?? "Untitled Event"
        title = decodeHTMLEntities(unfold(title))
        title = title.replacingOccurrences(of: #"\s+\d+$"#, with: "", options: .regularExpression)
            .trimmingCharacters(in: .whitespaces)

        let description = extractFoldedField("DESCRIPTION", from: block).map {
            decodeHTMLEntities(unfold($0))
        }
        let location = extractFoldedField("LOCATION", from: block).map { unfold($0) }

        let subject = extractSubject(from: title)
        let isAssignment = detectAssignment(title: title, subject: subject)
        let isMajor = detectMajor(title: title)

        let startDate = extractDate("DTSTART", from: block)
        let endDate = extractDate("DTEND", from: block)
        let dueDate = extractDate("DUE", from: block) ?? (isAssignment ? startDate : nil)
        let rrule = extractField("RRULE", from: block)

        return CalendarEvent(
            uid: uid,
            title: title,
            description: description,
            location: location,
            startDate: startDate,
            dueDate: dueDate,
            endDate: endDate,
            isAssignment: isAssignment,
            isMajor: isMajor,
            subject: subject,
            rrule: rrule,
            trinityFormat: subject != nil
        )
    }

    // MARK: - Field Extraction

    private static func extractField(_ name: String, from block: String) -> String? {
        let pattern = "\(name):(.+?)(?:\\r?\\n|$)"
        guard let regex = try? NSRegularExpression(pattern: pattern),
              let match = regex.firstMatch(in: block, range: NSRange(block.startIndex..., in: block)),
              let range = Range(match.range(at: 1), in: block) else { return nil }
        return String(block[range]).trimmingCharacters(in: .whitespacesAndNewlines)
    }

    private static func extractFoldedField(_ name: String, from block: String) -> String? {
        let pattern = "\(name):(.+?)(?=\\r?\\n[A-Z\\-]+:|$)"
        guard let regex = try? NSRegularExpression(pattern: pattern, options: [.dotMatchesLineSeparators]),
              let match = regex.firstMatch(in: block, range: NSRange(block.startIndex..., in: block)),
              let range = Range(match.range(at: 1), in: block) else { return nil }
        return String(block[range])
    }

    private static func extractDate(_ name: String, from block: String) -> Date? {
        let pattern = "\(name)(?:;[^:]*)?:(.+?)(?:\\r?\\n|$)"
        guard let regex = try? NSRegularExpression(pattern: pattern),
              let match = regex.firstMatch(in: block, range: NSRange(block.startIndex..., in: block)),
              let range = Range(match.range(at: 1), in: block) else { return nil }
        let raw = String(block[range]).trimmingCharacters(in: .whitespacesAndNewlines)
        return parseICalDate(raw)
    }

    // MARK: - Date Parsing

    static func parseICalDate(_ raw: String) -> Date? {
        let clean = raw.trimmingCharacters(in: .whitespacesAndNewlines)
        let formatter = DateFormatter()
        formatter.locale = Locale(identifier: "en_US_POSIX")

        // YYYYMMDDTHHMMSSZ
        if clean.count >= 15 && clean.contains("T") {
            formatter.dateFormat = clean.hasSuffix("Z") ? "yyyyMMdd'T'HHmmss'Z'" : "yyyyMMdd'T'HHmmss"
            if clean.hasSuffix("Z") { formatter.timeZone = TimeZone(identifier: "UTC") }
            return formatter.date(from: String(clean.prefix(15)))
        }
        // YYYYMMDD (date only)
        if clean.count == 8 {
            formatter.dateFormat = "yyyyMMdd"
            formatter.timeZone = TimeZone.current
            return formatter.date(from: clean)
        }
        return nil
    }

    // MARK: - Trinity-Specific Logic

    private static func extractSubject(from title: String) -> String? {
        let pattern = #"^((?:ADV\.\s+)?[A-Z][A-Z0-9\s:\/]+-\s*[A-Z0-9]+):"#
        guard let regex = try? NSRegularExpression(pattern: pattern),
              let match = regex.firstMatch(in: title, range: NSRange(title.startIndex..., in: title)),
              let range = Range(match.range(at: 1), in: title) else { return nil }
        return String(title[range]).trimmingCharacters(in: .whitespaces)
    }

    private static func detectAssignment(title: String, subject: String?) -> Bool {
        if subject != nil { return true }
        let keywords = #"\b(due|assignment|homework|test|quiz|exam|project|paper|lab|presentation|read|watch|complete|finish|study)\b"#
        return title.range(of: keywords, options: [.regularExpression, .caseInsensitive]) != nil
    }

    private static func detectMajor(title: String) -> Bool {
        let keywords = #"\b(test|quiz|exam|midterm|final|essay|paper|project|presentation|lab exam|oral exam)\b"#
        return title.range(of: keywords, options: [.regularExpression, .caseInsensitive]) != nil
    }

    // MARK: - Text Processing

    private static func unfold(_ s: String) -> String {
        s.replacingOccurrences(of: "\r\n ", with: "")
         .replacingOccurrences(of: "\r\n\t", with: "")
         .replacingOccurrences(of: "\n ", with: "")
         .replacingOccurrences(of: "\n\t", with: "")
    }

    private static func decodeHTMLEntities(_ s: String) -> String {
        var result = s
        // Numeric decimal entities
        if let regex = try? NSRegularExpression(pattern: "&#(\\d+);") {
            let matches = regex.matches(in: result, range: NSRange(result.startIndex..., in: result))
            for match in matches.reversed() {
                if let codeRange = Range(match.range(at: 1), in: result),
                   let code = UInt32(result[codeRange]),
                   let scalar = Unicode.Scalar(code) {
                    let fullRange = Range(match.range, in: result)!
                    result.replaceSubrange(fullRange, with: String(Character(scalar)))
                }
            }
        }
        // Named entities
        let named: [String: String] = [
            "&amp;": "&", "&lt;": "<", "&gt;": ">", "&quot;": "\"",
            "&#39;": "'", "&nbsp;": " ", "&ndash;": "–", "&mdash;": "—",
            "&ldquo;": "\u{201C}", "&rdquo;": "\u{201D}",
            "&lsquo;": "\u{2018}", "&rsquo;": "\u{2019}"
        ]
        for (entity, char) in named {
            result = result.replacingOccurrences(of: entity, with: char)
        }
        return result
    }
}

enum ICalError: Error {
    case invalidEncoding
    case networkError(Error)
}
```

```swift
import SwiftUI

extension Color {
    init(hex: String) {
        let hex = hex.trimmingCharacters(in: CharacterSet.alphanumerics.inverted)
        var int: UInt64 = 0
        Scanner(string: hex).scanHexInt64(&int)
        let r = Double((int >> 16) & 0xFF) / 255
        let g = Double((int >> 8) & 0xFF) / 255
        let b = Double(int & 0xFF) / 255
        self.init(red: r, green: g, blue: b)
    }

    var hexString: String {
        let nsColor = NSColor(self).usingColorSpace(.sRGB) ?? NSColor(self)
        let r = Int((nsColor.redComponent * 255).rounded())
        let g = Int((nsColor.greenComponent * 255).rounded())
        let b = Int((nsColor.blueComponent * 255).rounded())
        return String(format: "#%02X%02X%02X", r, g, b)
    }
}
```

```swift
import Foundation
import Combine

@MainActor
final class AppStore: ObservableObject {
    @Published var assignments: [Assignment] = []
    @Published var settings: AppSettings = AppSettings()
    @Published var isLoading: Bool = false
    @Published var lastError: String? = nil

    let storage: StorageService
    let themeManager: ThemeManager

    private let icalService = ICalService()
    private let notificationService: NotificationService
    private var refreshService: RefreshService?

    init(storage: StorageService = StorageService(), themeManager: ThemeManager = ThemeManager()) {
        self.storage = storage
        self.themeManager = themeManager
        self.notificationService = NotificationService()
        self.settings = storage.loadSettings()
        themeManager.apply(themeName: settings.themeName, glassStyle: settings.isGlassStyle)
    }

    func refresh() async {
        guard !settings.icalURL.isEmpty,
              let url = URL(string: settings.icalURL) else { return }
        isLoading = true
        lastError = nil
        do {
            let events = try await ICalService.fetch(url: url)
            assignments = storage.merge(events: events)
            notificationService.sync(assignments: assignments, settings: settings)
        } catch {
            lastError = error.localizedDescription
        }
        isLoading = false
    }

    func updateMetadata(_ metadata: EventMetadata, forUID uid: String) {
        storage.saveMetadata(metadata, forUID: uid)
        if let idx = assignments.firstIndex(where: { $0.id == uid }) {
            assignments[idx].metadata = metadata
        }
        notificationService.sync(assignments: assignments, settings: settings)
    }

    func saveSettings(_ newSettings: AppSettings) {
        settings = newSettings
        storage.saveSettings(newSettings)
        themeManager.apply(themeName: newSettings.themeName, glassStyle: newSettings.isGlassStyle)
        refreshService?.reschedule(interval: newSettings.refreshInterval)
    }

    func startBackgroundRefresh() {
        refreshService = RefreshService(interval: settings.refreshInterval) { [weak self] in
            Task { await self?.refresh() }
        }
        refreshService?.start()
    }

    // MARK: - Derived

    var pinnedAssignments: [Assignment] {
        assignments.filter { $0.isPinned && !$0.isCompleted }
    }

    var majorAssignments: [Assignment] {
        assignments.filter { $0.isMajor && !$0.isCompleted }
    }

    func assignments(for date: Date) -> [Assignment] {
        let cal = Calendar.current
        return assignments.filter { assignment in
            guard let due = assignment.dueDate else { return false }
            return cal.isDate(due, inSameDayAs: date)
        }.sorted { ($0.dueDate ?? .distantFuture) < ($1.dueDate ?? .distantFuture) }
    }

    func subjectColor(for subject: String?) -> Color {
        guard let subject else { return Color(hex: "#e2e8f0") }
        if let hex = settings.subjectColors[subject] { return Color(hex: hex) }
        let defaults = ["#e63946","#f77f00","#e9c46a","#2a9d8f","#118ab2","#8338ec","#ff006e","#8ac926","#06d6a0","#b56576","#3d405b","#ffb703"]
        let idx = abs(subject.hashValue) % defaults.count
        return Color(hex: defaults[idx])
    }
}
```

```swift
import Foundation

final class RefreshService {
    private var activity: NSBackgroundActivityScheduler?
    private var interval: TimeInterval
    private let handler: () -> Void

    init(interval: TimeInterval, handler: @escaping () -> Void) {
        self.interval = interval
        self.handler = handler
    }

    func start() {
        let scheduler = NSBackgroundActivityScheduler(identifier: "com.trintasks.refresh")
        scheduler.repeats = true
        scheduler.interval = interval
        scheduler.qualityOfService = .utility
        scheduler.schedule { [weak self] completion in
            self?.handler()
            completion(.success)
        }
        activity = scheduler
    }

    func reschedule(interval newInterval: TimeInterval) {
        activity?.invalidate()
        interval = newInterval
        start()
    }

    deinit { activity?.invalidate() }
}
```

```swift
import Foundation
import UserNotifications

final class NotificationService {
    private let center = UNUserNotificationCenter.current()

    func requestAuthorization() {
        center.requestAuthorization(options: [.alert, .sound, .badge]) { _, _ in }
    }

    func sync(assignments: [Assignment], settings: AppSettings) {
        center.getPendingNotificationRequests { [weak self] pending in
            guard let self else { return }
            let pendingIDs = Set(pending.map { $0.identifier })
            let now = Date()

            for assignment in assignments {
                guard !assignment.isCompleted,
                      let dueDate = assignment.dueDate,
                      let offset = assignment.metadata.reminderOffset else { continue }

                let fireDate = dueDate.addingTimeInterval(-offset)
                guard fireDate > now else { continue }

                let notifID = "reminder.\(assignment.id)"
                if pendingIDs.contains(notifID) { continue }

                let content = UNMutableNotificationContent()
                content.title = assignment.title
                content.body = "Due \(Self.formatDate(dueDate))"
                content.sound = .default

                let interval = fireDate.timeIntervalSinceNow
                let trigger = UNTimeIntervalNotificationTrigger(timeInterval: max(interval, 1), repeats: false)
                let request = UNNotificationRequest(identifier: notifID, content: content, trigger: trigger)
                self.center.add(request, withCompletionHandler: nil)
            }

            // Cancel notifications for completed or unpinned-reminder assignments
            let activeIDs = Set(assignments.compactMap { a -> String? in
                guard !a.isCompleted, a.metadata.reminderOffset != nil else { return nil }
                return "reminder.\(a.id)"
            })
            let toCancel = pendingIDs.filter { $0.hasPrefix("reminder.") && !activeIDs.contains($0) }
            self.center.removePendingNotificationRequests(withIdentifiers: Array(toCancel))
        }
    }

    private static func formatDate(_ date: Date) -> String {
        let f = DateFormatter()
        f.dateStyle = .short
        f.timeStyle = .short
        return f.string(from: date)
    }
}
```

```swift
        let panelSize = panel.frame.size

        let y = buttonRect.minY - panelSize.height - 4
        let origin = NSPoint(
            x: max(screen.visibleFrame.minX, min(x, screen.visibleFrame.maxX - panelSize.width)),
            y: max(screen.visibleFrame.minY, y)
        )
        panel.setFrameOrigin(origin)
    }
```

```swift
            headerButton(icon: "gearshape", action: { showSettings = true })
        }
        .padding(.horizontal, 12)

        .border(themeManager.current.borderColor, width: BrutalStyle.borderWidth)
    }

    private func headerButton(icon: String, action: @escaping () -> Void) -> some View {
        Button(action: action) {
            Image(systemName: icon)
                .font(.system(size: 13, weight: .semibold))
                .foregroundColor(themeManager.current.accentContrast)
                .frame(width: 28, height: 28)
                .brutalCard(fill: themeManager.current.accent, border: themeManager.current.accentContrast.opacity(0.4))
        }
        .buttonStyle(.plain)
    }

    private func saveURL() {
        var s = store.settings
        s.icalURL = urlInput
        store.saveSettings(s)
        Task { await store.refresh() }
    }

    private func goToToday() {
        weekStart = Calendar.current.startOfWeek(for: Date())
    }

    private func shiftWeek(_ delta: Int) {
        weekStart = Calendar.current.date(byAdding: .weekOfYear, value: delta, to: weekStart) ?? weekStart
    }
}

private extension Calendar {
    func startOfWeek(for date: Date) -> Date {
        let comps = dateComponents([.yearForWeekOfYear, .weekOfYear], from: date)
        return self.date(from: comps) ?? date
    }
}
```

```bash
#!/bin/bash
set -e

APP_NAME="TrinTasks"
SCHEME="TrinTasks"
ARCHIVE_PATH="build/${APP_NAME}.xcarchive"
EXPORT_PATH="build/export"
DMG_PATH="build/${APP_NAME}.dmg"

echo "Archiving..."
xcodebuild archive \
  -project "TrinTasksApp/${APP_NAME}.xcodeproj" \
  -scheme "${SCHEME}" \
  -configuration Release \
  -archivePath "${ARCHIVE_PATH}"

echo "Exporting..."
xcodebuild -exportArchive \
  -archivePath "${ARCHIVE_PATH}" \
  -exportPath "${EXPORT_PATH}" \
  -exportOptionsPlist scripts/ExportOptions.plist

echo "Creating DMG..."
create-dmg \
  --volname "${APP_NAME}" \
  --window-size 500 320 \
  --icon-size 100 \
  --icon "${APP_NAME}.app" 120 160 \
  --app-drop-link 380 160 \
  "${DMG_PATH}" \
  "${EXPORT_PATH}/${APP_NAME}.app"

echo "Done: ${DMG_PATH}"
```
